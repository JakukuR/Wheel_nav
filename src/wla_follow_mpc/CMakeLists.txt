cmake_minimum_required(VERSION 3.16)
project(wla_follow_mpc)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(wla_diff_mpc REQUIRED)
foreach(dep rclcpp nav_msgs geometry_msgs tf2 tf2_geometry_msgs)
  find_package(${dep} REQUIRED)
endforeach()
rosidl_generate_interfaces(${PROJECT_NAME} "srv/ComputeMpc.srv" DEPENDENCIES nav_msgs geometry_msgs)
add_executable(follow_mpc_server src/server.cpp)
target_compile_features(follow_mpc_server PRIVATE cxx_std_17)
target_link_libraries(follow_mpc_server wla_diff_mpc::wla_diff_mpc_core)
set_target_properties(follow_mpc_server PROPERTIES INSTALL_RPATH_USE_LINK_PATH TRUE)
ament_target_dependencies(follow_mpc_server rclcpp nav_msgs geometry_msgs tf2 tf2_geometry_msgs)
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} "rosidl_typesupport_cpp")
target_link_libraries(follow_mpc_server "${typesupport}")
install(TARGETS follow_mpc_server DESTINATION lib/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
ament_package()
