cmake_minimum_required(VERSION 3.16)
project(wla_follow_local)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development)
find_package(pybind11 REQUIRED)
find_package(Eigen3 REQUIRED)
pybind11_add_module(_wla_follow_local src/local.cpp)
target_compile_features(_wla_follow_local PRIVATE cxx_std_17)
target_compile_options(_wla_follow_local PRIVATE -O2 -Wall -Wextra -Wpedantic)
target_link_libraries(_wla_follow_local PRIVATE Eigen3::Eigen)
ament_get_python_install_dir(PYTHON_INSTALL_DIR)
install(TARGETS _wla_follow_local LIBRARY DESTINATION "${PYTHON_INSTALL_DIR}")
ament_package()
