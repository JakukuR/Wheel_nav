cmake_minimum_required(VERSION 3.20)
project(wla_uvc_depth LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc calib3d imgcodecs)
find_package(yaml-cpp REQUIRED)
find_package(nlohmann_json REQUIRED)
include(CTest)
add_library(uvc_options INTERFACE)
target_compile_features(uvc_options INTERFACE cxx_std_20)
target_compile_options(uvc_options INTERFACE -Wall -Wextra -Wpedantic -ffp-contract=off
  "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=wla_uvc_depth"
  $<$<CONFIG:Debug>:-O0;-g3> $<$<CONFIG:Release>:-O2;-fno-omit-frame-pointer>
  $<$<CONFIG:RelWithDebInfo>:-O2;-fno-omit-frame-pointer>)
add_library(uvc_depth_core src/core.cc)
target_include_directories(uvc_depth_core PUBLIC include)
target_include_directories(uvc_depth_core SYSTEM PUBLIC ${OpenCV_INCLUDE_DIRS})
target_link_libraries(uvc_depth_core PUBLIC uvc_options ${OpenCV_LIBS} yaml-cpp)
target_compile_options(uvc_depth_core PRIVATE -Werror)
add_executable(uvc_depth_cpp src/node.cc)
target_link_libraries(uvc_depth_cpp uvc_depth_core nlohmann_json::nlohmann_json)
ament_target_dependencies(uvc_depth_cpp rclcpp rcl_interfaces sensor_msgs std_msgs)
if(BUILD_TESTING)
  add_executable(test_geometry test/core_test.cc)
  target_link_libraries(test_geometry uvc_depth_core)
  target_compile_definitions(test_geometry PRIVATE FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/test/rig.yaml")
  add_test(NAME geometry_matching_projection COMMAND test_geometry)
  add_executable(compare_reference test/compare_reference.cc)
  target_link_libraries(compare_reference uvc_depth_core)
endif()
install(TARGETS uvc_depth_cpp DESTINATION lib/${PROJECT_NAME})
add_library(snapshot_field_core src/snapshot_field.cc)
target_include_directories(snapshot_field_core PUBLIC include)
target_include_directories(snapshot_field_core SYSTEM PUBLIC ${OpenCV_INCLUDE_DIRS})
target_link_libraries(snapshot_field_core PUBLIC uvc_options ${OpenCV_LIBS})
target_compile_options(snapshot_field_core PRIVATE -Werror)
add_executable(stereo_snapshot_field src/snapshot_field_node.cc)
target_link_libraries(stereo_snapshot_field snapshot_field_core nlohmann_json::nlohmann_json)
ament_target_dependencies(stereo_snapshot_field rclcpp sensor_msgs std_msgs)
install(TARGETS stereo_snapshot_field DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  add_executable(snapshot_field_test test/snapshot_field_test.cc)
  target_link_libraries(snapshot_field_test snapshot_field_core)
  add_test(NAME snapshot_field_visibility_and_distance COMMAND snapshot_field_test)
endif()
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(geometry_msgs REQUIRED)
add_library(ground_filter_core src/ground_filter.cc)
target_include_directories(ground_filter_core PUBLIC include)
target_include_directories(ground_filter_core SYSTEM PUBLIC ${OpenCV_INCLUDE_DIRS})
target_link_libraries(ground_filter_core PUBLIC uvc_options ${OpenCV_LIBS})
add_executable(stereo_ground_filter src/ground_filter_node.cc)
target_link_libraries(stereo_ground_filter ground_filter_core nlohmann_json::nlohmann_json)
ament_target_dependencies(stereo_ground_filter rclcpp sensor_msgs std_msgs geometry_msgs tf2 tf2_ros)
install(TARGETS stereo_ground_filter DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  add_executable(ground_filter_test test/ground_filter_test.cc)
  target_link_libraries(ground_filter_test ground_filter_core)
  add_test(NAME ground_filter_classification COMMAND ground_filter_test)
endif()
option(WLA_BUILD_STEREO_TRT "Build native calibrated TensorRT stereo experiment" OFF)
if(WLA_BUILD_STEREO_TRT)
  find_package(CUDAToolkit REQUIRED)
  find_path(TENSORRT_INCLUDE_DIR NvInfer.h PATH_SUFFIXES aarch64-linux-gnu REQUIRED)
  find_library(TENSORRT_LIBRARY nvinfer REQUIRED)
  find_package(tf2 REQUIRED)
  find_package(tf2_ros REQUIRED)
  find_package(geometry_msgs REQUIRED)
  enable_language(CUDA)
  add_library(stereo_cuda src/stereo_cuda.cu)
  set_target_properties(stereo_cuda PROPERTIES CUDA_ARCHITECTURES 87 CUDA_STANDARD 17)
  target_link_libraries(stereo_cuda PUBLIC CUDA::cudart)
  add_library(stereo_engine src/stereo_engine.cc)
  target_include_directories(stereo_engine PUBLIC include)
  target_include_directories(stereo_engine SYSTEM PUBLIC ${OpenCV_INCLUDE_DIRS} ${TENSORRT_INCLUDE_DIR})
  target_link_libraries(stereo_engine PUBLIC uvc_options ${OpenCV_LIBS} CUDA::cudart stereo_cuda ${TENSORRT_LIBRARY} ${CMAKE_DL_LIBS})
  target_compile_options(stereo_engine PRIVATE -Werror)
  if(BUILD_TESTING)
    add_executable(stereo_input_test test/stereo_input_test.cc)
    target_link_libraries(stereo_input_test stereo_engine)
    add_test(NAME stereo_input_contract COMMAND stereo_input_test)
    add_executable(stereo_gpu_geometry_test test/stereo_gpu_geometry_test.cc)
    target_link_libraries(stereo_gpu_geometry_test stereo_cuda)
    add_test(NAME stereo_gpu_invalid_geometry COMMAND stereo_gpu_geometry_test)
  endif()
  add_executable(packed_pipeline_test test/packed_pipeline_test.cc src/direct_capture.cc)
  target_link_libraries(packed_pipeline_test stereo_engine uvc_depth_core nlohmann_json::nlohmann_json)
  add_executable(stereo_probe test/stereo_probe.cc)
  target_link_libraries(stereo_probe stereo_engine uvc_depth_core nlohmann_json::nlohmann_json)
  add_executable(uvc_stereo_trt src/stereo_node.cc src/direct_capture.cc)
  target_link_libraries(uvc_stereo_trt stereo_engine uvc_depth_core nlohmann_json::nlohmann_json)
  ament_target_dependencies(uvc_stereo_trt rclcpp sensor_msgs std_msgs geometry_msgs tf2 tf2_ros)
  install(TARGETS uvc_stereo_trt stereo_probe DESTINATION lib/${PROJECT_NAME})
endif()
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_package()
