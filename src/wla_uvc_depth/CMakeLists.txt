cmake_minimum_required(VERSION 3.20)
project(wla_uvc_depth LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc calib3d imgcodecs)
find_package(yaml-cpp REQUIRED)
find_package(nlohmann_json REQUIRED)
include(CTest)
add_library(uvc_options INTERFACE)
target_compile_features(uvc_options INTERFACE cxx_std_20)
target_compile_options(uvc_options INTERFACE -Wall -Wextra -Wpedantic -ffp-contract=off
  "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=wla_uvc_depth"
  $<$<CONFIG:Debug>:-O0;-g3> $<$<CONFIG:Release>:-O2;-fno-omit-frame-pointer>
  $<$<CONFIG:RelWithDebInfo>:-O2;-fno-omit-frame-pointer>)
add_library(uvc_depth_core src/core.cc)
target_include_directories(uvc_depth_core PUBLIC include)
target_include_directories(uvc_depth_core SYSTEM PUBLIC ${OpenCV_INCLUDE_DIRS})
target_link_libraries(uvc_depth_core PUBLIC uvc_options ${OpenCV_LIBS} yaml-cpp)
target_compile_options(uvc_depth_core PRIVATE -Werror)
add_executable(uvc_depth_cpp src/node.cc)
target_link_libraries(uvc_depth_cpp uvc_depth_core nlohmann_json::nlohmann_json)
ament_target_dependencies(uvc_depth_cpp rclcpp rcl_interfaces sensor_msgs std_msgs)
if(BUILD_TESTING)
  add_executable(test_geometry test/core_test.cc)
  target_link_libraries(test_geometry uvc_depth_core)
  target_compile_definitions(test_geometry PRIVATE FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/test/rig.yaml")
  add_test(NAME geometry_matching_projection COMMAND test_geometry)
  add_executable(compare_reference test/compare_reference.cc)
  target_link_libraries(compare_reference uvc_depth_core)
endif()
install(TARGETS uvc_depth_cpp DESTINATION lib/${PROJECT_NAME})
option(WLA_BUILD_STEREO_TRT "Build native calibrated TensorRT stereo experiment" OFF)
if(WLA_BUILD_STEREO_TRT)
  find_package(CUDAToolkit REQUIRED)
  find_path(TENSORRT_INCLUDE_DIR NvInfer.h PATH_SUFFIXES aarch64-linux-gnu REQUIRED)
  find_library(TENSORRT_LIBRARY nvinfer REQUIRED)
  find_package(tf2 REQUIRED)
  find_package(tf2_ros REQUIRED)
  find_package(geometry_msgs REQUIRED)
  add_library(stereo_engine src/stereo_engine.cc)
  target_include_directories(stereo_engine PUBLIC include)
  target_include_directories(stereo_engine SYSTEM PUBLIC ${OpenCV_INCLUDE_DIRS} ${TENSORRT_INCLUDE_DIR})
  target_link_libraries(stereo_engine PUBLIC uvc_options ${OpenCV_LIBS} CUDA::cudart ${TENSORRT_LIBRARY})
  target_compile_options(stereo_engine PRIVATE -Werror)
  if(BUILD_TESTING)
    add_executable(stereo_input_test test/stereo_input_test.cc)
    target_link_libraries(stereo_input_test stereo_engine)
    add_test(NAME stereo_input_contract COMMAND stereo_input_test)
  endif()
  add_executable(stereo_probe test/stereo_probe.cc)
  target_link_libraries(stereo_probe stereo_engine uvc_depth_core nlohmann_json::nlohmann_json)
  add_executable(uvc_stereo_trt src/stereo_node.cc)
  target_link_libraries(uvc_stereo_trt stereo_engine uvc_depth_core nlohmann_json::nlohmann_json)
  ament_target_dependencies(uvc_stereo_trt rclcpp sensor_msgs std_msgs geometry_msgs tf2 tf2_ros)
  install(TARGETS uvc_stereo_trt stereo_probe DESTINATION lib/${PROJECT_NAME})
endif()
ament_package()
