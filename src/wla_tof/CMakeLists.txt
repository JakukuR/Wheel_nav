cmake_minimum_required(VERSION 3.20)
project(wla_tof LANGUAGES CXX)
option(WLA_TOF_BUILD_ROS "Build the ROS serial receiver" ON)
include(CTest)
add_library(wla_tof_options INTERFACE)
target_compile_features(wla_tof_options INTERFACE cxx_std_20)
target_compile_options(wla_tof_options INTERFACE -Wall -Wextra -Wpedantic -ffp-contract=off
  "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=wla_tof"
  $<$<CONFIG:Debug>:-O0;-g3>
  $<$<CONFIG:Release>:-O2;-fno-omit-frame-pointer>
  $<$<CONFIG:RelWithDebInfo>:-O2;-fno-omit-frame-pointer>)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(nlohmann_json REQUIRED)
add_library(wla_tof_core src/protocol.cc src/geometry.cc)
target_include_directories(wla_tof_core PUBLIC include)
target_link_libraries(wla_tof_core PUBLIC wla_tof_options Eigen3::Eigen nlohmann_json::nlohmann_json)
target_compile_options(wla_tof_core PRIVATE -Werror)
if(BUILD_TESTING)
  add_executable(test_protocol test/test_protocol.cc)
  target_link_libraries(test_protocol wla_tof_core)
  add_test(NAME protocol_and_projection COMMAND test_protocol)
  add_executable(test_geometry test/test_geometry.cc)
  target_link_libraries(test_geometry wla_tof_core)
  target_compile_definitions(test_geometry PRIVATE GEOMETRY_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/test/robot_geometry.json")
  add_test(NAME joint_ground_geometry COMMAND test_geometry)
endif()
if(WLA_TOF_BUILD_ROS)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(geometry_msgs REQUIRED)
  find_package(tf2_ros REQUIRED)
  find_package(nlohmann_json REQUIRED)
  add_executable(tof_serial src/tof_node.cc)
  target_link_libraries(tof_serial wla_tof_core nlohmann_json::nlohmann_json)
  target_compile_options(tof_serial PRIVATE -Werror)
  ament_target_dependencies(tof_serial rclcpp sensor_msgs std_msgs geometry_msgs tf2_ros)
  add_executable(tof_obstacles src/obstacle_node.cc)
  target_link_libraries(tof_obstacles wla_tof_core)
  target_compile_options(tof_obstacles PRIVATE -Werror)
  ament_target_dependencies(tof_obstacles rclcpp sensor_msgs std_msgs)
  install(TARGETS tof_serial tof_obstacles DESTINATION lib/${PROJECT_NAME})
  ament_package()
endif()
