cmake_minimum_required(VERSION 3.16)
project(wla_cuvslam_validation)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(Eigen3 REQUIRED)
set(CUVSLAM_SOURCE_DIR "" CACHE PATH "Pinned cuVSLAM source root")
set(CUVSLAM_BUILD_DIR "" CACHE PATH "cuVSLAM SDK build root")
find_path(CUVSLAM_INCLUDE_DIR cuvslam2.h HINTS "${CUVSLAM_SOURCE_DIR}/libs/cuvslam" REQUIRED)
find_library(CUVSLAM_LIBRARY cuvslam HINTS "${CUVSLAM_BUILD_DIR}/bin" REQUIRED NO_DEFAULT_PATH)
add_executable(cuvslam_validation src/cuvslam_validation.cpp)
target_include_directories(cuvslam_validation PRIVATE ${CUVSLAM_INCLUDE_DIR})
target_link_libraries(cuvslam_validation ${CUVSLAM_LIBRARY} Eigen3::Eigen)
ament_target_dependencies(cuvslam_validation rclcpp sensor_msgs nav_msgs std_msgs tf2_ros)
target_compile_options(cuvslam_validation PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(cuvslam_validation PROPERTIES INSTALL_RPATH "${CUVSLAM_BUILD_DIR}/bin")
install(TARGETS cuvslam_validation DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/sensor_probe.py scripts/run_validation.py scripts/replay_validation.py scripts/check_input_guards.py DESTINATION lib/${PROJECT_NAME})
ament_package()
