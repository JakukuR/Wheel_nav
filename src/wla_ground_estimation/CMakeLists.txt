cmake_minimum_required(VERSION 3.20)
project(wla_ground_estimation LANGUAGES C CXX)
option(WLA_GROUND_BUILD_ROS "Build the ROS 2 transport and interfaces" ON)
include(CTest)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(tinyxml2 CONFIG QUIET)
if(NOT TARGET tinyxml2::tinyxml2)
  find_path(TINYXML2_INCLUDE_DIR tinyxml2.h REQUIRED)
  find_library(TINYXML2_LIBRARY NAMES tinyxml2 REQUIRED)
  add_library(tinyxml2::tinyxml2 UNKNOWN IMPORTED)
  set_target_properties(tinyxml2::tinyxml2 PROPERTIES IMPORTED_LOCATION "${TINYXML2_LIBRARY}"
    INTERFACE_INCLUDE_DIRECTORIES "${TINYXML2_INCLUDE_DIR}")
endif()
add_library(wla_ground_options INTERFACE)
target_compile_features(wla_ground_options INTERFACE cxx_std_20)
target_compile_options(wla_ground_options INTERFACE -Wall -Wextra -Wpedantic -ffp-contract=off
  "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=wla_ground_estimation"
  $<$<CONFIG:Debug>:-O0;-g3>
  $<$<CONFIG:Release>:-O2;-fno-omit-frame-pointer>
  $<$<CONFIG:RelWithDebInfo>:-O2;-fno-omit-frame-pointer>)
add_library(wla_ground_core src/geometry.cc)
target_include_directories(wla_ground_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(wla_ground_core PUBLIC Eigen3::Eigen wla_ground_options PRIVATE tinyxml2::tinyxml2)
target_compile_options(wla_ground_core PRIVATE -Werror)
if(BUILD_TESTING)
  foreach(name geometry time)
    add_executable(test_ground_${name} test/cpp/test_${name}.cc)
    target_link_libraries(test_ground_${name} PRIVATE wla_ground_core)
    target_compile_definitions(test_ground_${name} PRIVATE
      TEST_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/test/data")
    add_test(NAME ground_${name} COMMAND test_ground_${name})
  endforeach()
endif()
if(WLA_GROUND_BUILD_ROS)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(geometry_msgs REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(rosidl_default_generators REQUIRED)
  find_package(nlohmann_json REQUIRED)
  rosidl_generate_interfaces(${PROJECT_NAME} "msg/GroundEstimate.msg"
    DEPENDENCIES std_msgs geometry_msgs)
  add_executable(ground_estimator src/ground_node.cc)
  ament_target_dependencies(ground_estimator rclcpp sensor_msgs geometry_msgs std_msgs)
  rosidl_get_typesupport_target(ground_typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
  target_link_libraries(ground_estimator wla_ground_core nlohmann_json::nlohmann_json "${ground_typesupport}")
  install(TARGETS ground_estimator DESTINATION lib/${PROJECT_NAME})
  install(TARGETS wla_ground_core ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
  install(DIRECTORY include/ DESTINATION include)
  ament_export_dependencies(rosidl_default_runtime)
  ament_package()
endif()
