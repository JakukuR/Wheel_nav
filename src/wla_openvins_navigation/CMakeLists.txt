cmake_minimum_required(VERSION 3.16)
project(wla_openvins_navigation)
include(CTest)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED)
find_package(Boost REQUIRED COMPONENTS system filesystem thread date_time)
foreach(dep rclcpp std_msgs std_srvs builtin_interfaces geometry_msgs nav_msgs sensor_msgs tf2_ros ov_core ov_init ov_msckf)
  find_package(${dep} REQUIRED)
endforeach()
rosidl_generate_interfaces(${PROJECT_NAME} "msg/EstimatorStatus.msg" "msg/ImuDiagnostics.msg" DEPENDENCIES std_msgs builtin_interfaces)
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)
add_executable(openvins_frontend src/frontend.cpp)
target_compile_definitions(openvins_frontend PRIVATE ROS_AVAILABLE=2)
target_include_directories(openvins_frontend PRIVATE include ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(openvins_frontend rclcpp std_msgs std_srvs geometry_msgs nav_msgs sensor_msgs tf2_ros ov_core ov_init ov_msckf)
target_link_libraries(openvins_frontend Eigen3::Eigen ${OpenCV_LIBRARIES} ${Boost_LIBRARIES} "${cpp_typesupport_target}")
install(TARGETS openvins_frontend DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config patches DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/generate_config.py scripts/inspect_config.py scripts/quality_gate.py scripts/record_metrics.py scripts/stereo_fault_injector.py DESTINATION lib/${PROJECT_NAME})
install(FILES README.md COPYING VERSION.json openvins.repos DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  enable_testing()
  add_test(NAME calibration_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_calibration.py)
  add_executable(test_adapter test/test_adapter.cpp)
  target_include_directories(test_adapter PRIVATE include)
  target_link_libraries(test_adapter Eigen3::Eigen)
  target_compile_options(test_adapter PRIVATE -UNDEBUG)
  add_test(NAME adapter_contract COMMAND test_adapter)
  add_executable(test_inertial test/test_inertial.cpp)
  target_compile_definitions(test_inertial PRIVATE ROS_AVAILABLE=2)
  target_compile_options(test_inertial PRIVATE -UNDEBUG)
  target_include_directories(test_inertial PRIVATE include ${OpenCV_INCLUDE_DIRS})
  ament_target_dependencies(test_inertial rclcpp ov_core ov_init ov_msckf)
  target_link_libraries(test_inertial Eigen3::Eigen ${OpenCV_LIBRARIES} ${Boost_LIBRARIES})
  add_test(NAME inertial_continuity COMMAND test_inertial ${CMAKE_CURRENT_SOURCE_DIR}/config/synthetic/estimator_config.yaml)
  add_executable(test_zupt test/test_zupt.cpp)
  target_compile_definitions(test_zupt PRIVATE ROS_AVAILABLE=2)
  target_compile_options(test_zupt PRIVATE -UNDEBUG)
  target_include_directories(test_zupt PRIVATE include ${OpenCV_INCLUDE_DIRS})
  ament_target_dependencies(test_zupt rclcpp ov_core ov_init ov_msckf)
  target_link_libraries(test_zupt Eigen3::Eigen ${OpenCV_LIBRARIES} ${Boost_LIBRARIES})
  add_test(NAME zupt_constraints COMMAND test_zupt ${CMAKE_CURRENT_SOURCE_DIR}/config/synthetic/estimator_config.yaml)
  add_test(NAME sparse_visual_updates COMMAND test_inertial ${CMAKE_CURRENT_SOURCE_DIR}/config/synthetic/estimator_config.yaml --sparse-updates)
endif()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
