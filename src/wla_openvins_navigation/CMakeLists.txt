cmake_minimum_required(VERSION 3.16)
project(wla_openvins_navigation)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
add_executable(odometry_adapter src/odometry_adapter.cpp)
target_include_directories(odometry_adapter PRIVATE include)
ament_target_dependencies(odometry_adapter rclcpp nav_msgs geometry_msgs tf2_ros)
target_link_libraries(odometry_adapter Eigen3::Eigen)
install(TARGETS odometry_adapter DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config patches DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/generate_config.py scripts/inspect_config.py scripts/record_metrics.py
  scripts/stereo_fault_injector.py DESTINATION lib/${PROJECT_NAME})
install(FILES README.md COPYING VERSION.json openvins.repos DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  enable_testing()
  add_test(NAME calibration_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_calibration.py)
  add_executable(test_adapter test/test_adapter.cpp)
  target_include_directories(test_adapter PRIVATE include)
  target_link_libraries(test_adapter Eigen3::Eigen)
  target_compile_options(test_adapter PRIVATE -UNDEBUG)
  add_test(NAME adapter_contract COMMAND test_adapter)
endif()
ament_package()
