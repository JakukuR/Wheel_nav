cmake_minimum_required(VERSION 3.16)
project(wla_cuvslam_navigation)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2 REQUIRED)
find_package(Eigen3 REQUIRED)
set(CUVSLAM_SOURCE_DIR "" CACHE PATH "Pinned cuVSLAM v17 source")
set(CUVSLAM_BUILD_DIR "" CACHE PATH "Pinned cuVSLAM SDK build")
find_path(CUVSLAM_INCLUDE_DIR cuvslam2.h HINTS "${CUVSLAM_SOURCE_DIR}/libs/cuvslam" REQUIRED)
find_library(CUVSLAM_LIBRARY cuvslam HINTS "${CUVSLAM_BUILD_DIR}/bin" REQUIRED NO_DEFAULT_PATH)
add_library(upstream_pose_cache STATIC third_party/isaac_ros_cuvslam/pose_cache.cpp)
target_include_directories(upstream_pose_cache PUBLIC third_party/isaac_ros_cuvslam/include)
ament_target_dependencies(upstream_pose_cache tf2)
add_executable(cuvslam_odometry src/cuvslam_odometry.cpp)
target_include_directories(cuvslam_odometry PRIVATE ${CUVSLAM_INCLUDE_DIR} src)
target_link_libraries(cuvslam_odometry ${CUVSLAM_LIBRARY} Eigen3::Eigen upstream_pose_cache)
ament_target_dependencies(cuvslam_odometry rclcpp sensor_msgs nav_msgs std_msgs geometry_msgs tf2_ros)
target_compile_options(cuvslam_odometry PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(cuvslam_odometry PROPERTIES INSTALL_RPATH "${CUVSLAM_BUILD_DIR}/bin")
install(TARGETS cuvslam_odometry DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_feature_overlay test/test_feature_overlay.cpp)
  target_include_directories(test_feature_overlay PRIVATE src)
  target_compile_options(test_feature_overlay PRIVATE -UNDEBUG)
  add_test(NAME feature_overlay COMMAND test_feature_overlay)
  add_executable(test_feature_preview_ros test/test_feature_preview_ros.cpp)
  target_include_directories(test_feature_preview_ros PRIVATE src)
  ament_target_dependencies(test_feature_preview_ros rclcpp sensor_msgs)
  target_compile_options(test_feature_preview_ros PRIVATE -UNDEBUG)
  add_test(NAME feature_preview_ros COMMAND test_feature_preview_ros)
  set_tests_properties(feature_preview_ros PROPERTIES
    ENVIRONMENT "ROS_DOMAIN_ID=198;ROS_AUTOMATIC_DISCOVERY_RANGE=LOCALHOST;RMW_IMPLEMENTATION=rmw_cyclonedds_cpp"
    RUN_SERIAL TRUE TIMEOUT 15)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/test_upstream_frame_recovery.cpp
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_upstream_frame_recovery.py
      ${CMAKE_CURRENT_SOURCE_DIR}/src/cuvslam_odometry.cpp
      ${CMAKE_CURRENT_BINARY_DIR}/test_upstream_frame_recovery.cpp
    DEPENDS src/cuvslam_odometry.cpp test/test_upstream_frame_recovery.py)
  add_executable(test_upstream_frame_recovery ${CMAKE_CURRENT_BINARY_DIR}/test_upstream_frame_recovery.cpp)
  target_link_libraries(test_upstream_frame_recovery upstream_pose_cache Eigen3::Eigen)
  target_compile_options(test_upstream_frame_recovery PRIVATE -UNDEBUG)
  add_test(NAME upstream_frame_recovery COMMAND test_upstream_frame_recovery)
  add_executable(test_visual_pose_evidence test/test_visual_pose_evidence.cpp)
  target_include_directories(test_visual_pose_evidence PRIVATE src)
  target_compile_options(test_visual_pose_evidence PRIVATE -UNDEBUG)
  add_test(NAME visual_pose_evidence COMMAND test_visual_pose_evidence)
  add_executable(test_health_evidence test/test_health_evidence.cpp)
  target_include_directories(test_health_evidence PRIVATE src)
  add_test(NAME health_evidence COMMAND test_health_evidence)
  add_executable(test_odometry_contract test/test_odometry_contract.cpp)
  target_include_directories(test_odometry_contract PRIVATE src)
  target_link_libraries(test_odometry_contract Eigen3::Eigen)
  add_test(NAME odometry_contract COMMAND test_odometry_contract)
  add_executable(test_initialization_sampling test/test_initialization_sampling.cpp)
  target_include_directories(test_initialization_sampling PRIVATE src)
  target_compile_options(test_initialization_sampling PRIVATE -UNDEBUG)
  add_test(NAME initialization_sampling COMMAND test_initialization_sampling)
endif()
ament_package()
