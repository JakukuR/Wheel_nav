cmake_minimum_required(VERSION 3.16)
project(wla_follow_transport)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(wla_behavior_interfaces REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(follow_transport src/follow_transport.cpp)
target_include_directories(follow_transport PRIVATE include)
target_compile_features(follow_transport PRIVATE cxx_std_17)
target_compile_options(follow_transport PRIVATE -O2 -Wall -Wextra -Wpedantic)
target_link_libraries(follow_transport nlohmann_json::nlohmann_json)
ament_target_dependencies(follow_transport rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs std_srvs tf2_ros wla_behavior_interfaces)
install(TARGETS follow_transport DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_transport_policy test/test_transport_policy.cpp)
  target_include_directories(test_transport_policy PRIVATE include)
  target_compile_features(test_transport_policy PRIVATE cxx_std_17)
  target_link_libraries(test_transport_policy nlohmann_json::nlohmann_json)
  add_test(NAME transport_policy COMMAND test_transport_policy)
endif()
ament_package()
