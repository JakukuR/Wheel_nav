cmake_minimum_required(VERSION 3.20)
project(wla_robot_state LANGUAGES CXX)
option(WLA_STATE_BUILD_ROS "Build ROS state adapter" ON)
include(CTest)
find_package(Eigen3 REQUIRED NO_MODULE)
find_package(nlohmann_json REQUIRED)
add_library(wla_state_options INTERFACE)
target_compile_features(wla_state_options INTERFACE cxx_std_20)
target_compile_options(wla_state_options INTERFACE -Wall -Wextra -Wpedantic -ffp-contract=off
  "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=wla_robot_state"
  $<$<CONFIG:Debug>:-O0;-g3>
  $<$<CONFIG:Release>:-O2;-fno-omit-frame-pointer>
  $<$<CONFIG:RelWithDebInfo>:-O2;-fno-omit-frame-pointer>)
add_library(wla_state_core src/state.cc)
target_include_directories(wla_state_core PUBLIC include)
target_link_libraries(wla_state_core PUBLIC wla_state_options Eigen3::Eigen nlohmann_json::nlohmann_json)
target_compile_options(wla_state_core PRIVATE -Werror)
if(BUILD_TESTING)
  add_executable(test_state test/test_state.cc)
  target_link_libraries(test_state wla_state_core)
  add_test(NAME state_semantics COMMAND test_state)
endif()
if(WLA_STATE_BUILD_ROS)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(geometry_msgs REQUIRED)
  find_package(std_msgs REQUIRED)
  add_executable(robot_state_adapter src/state_node.cc)
  target_link_libraries(robot_state_adapter wla_state_core)
  ament_target_dependencies(robot_state_adapter rclcpp sensor_msgs geometry_msgs std_msgs)
  install(TARGETS robot_state_adapter DESTINATION lib/${PROJECT_NAME})
  ament_package()
endif()
