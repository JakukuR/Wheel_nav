cmake_minimum_required(VERSION 3.16)
project(wla_behavior_bridge)
find_package(ament_cmake REQUIRED)
set(WLA_AGENT_SOURCE_DIR "$ENV{HOME}/Wheel_Legged_Agent" CACHE PATH "Agent source repository supplying the shared behavior core")
if(NOT EXISTS "${WLA_AGENT_SOURCE_DIR}/src/wla_sim/behavior_bridge/agent_semantics.py")
  message(FATAL_ERROR "Set WLA_AGENT_SOURCE_DIR to the integrated Agent repository")
endif()
ament_environment_hooks("${CMAKE_CURRENT_SOURCE_DIR}/env-hooks/agent_runtime.dsv.in")
install(PROGRAMS scripts/navigation_task_gateway scripts/task_velocity_gate scripts/agent_semantic_api scripts/nav_perception_adapter scripts/agent_person_api DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_package()
