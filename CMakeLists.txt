cmake_minimum_required(VERSION 3.16)
project(wla_r680_navigation)

if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(lifecycle_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rtabmap_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)

add_executable(command_guard src/command_guard.cpp)
target_include_directories(command_guard PRIVATE include)
ament_target_dependencies(command_guard geometry_msgs rclcpp std_msgs)
add_executable(vio_initializer src/vio_initializer.cpp)
target_include_directories(vio_initializer PRIVATE include)
ament_target_dependencies(vio_initializer geometry_msgs lifecycle_msgs nav_msgs rclcpp sensor_msgs std_msgs)
add_executable(mapping_odom_gate src/mapping_odom_gate.cpp)
ament_target_dependencies(mapping_odom_gate nav_msgs rclcpp std_msgs)

add_executable(imu_conditioner src/imu_conditioner.cpp)
add_executable(path_speed_profile src/path_speed_profile.cpp)
add_executable(goal_approach_limiter src/goal_approach_limiter.cpp)
add_executable(dynamic_obstacle_memory src/dynamic_obstacle_memory.cpp)
ament_target_dependencies(imu_conditioner rclcpp sensor_msgs)
ament_target_dependencies(path_speed_profile nav2_msgs nav_msgs rclcpp std_msgs)
ament_target_dependencies(goal_approach_limiter geometry_msgs nav_msgs rclcpp std_msgs tf2 tf2_ros)
ament_target_dependencies(dynamic_obstacle_memory geometry_msgs rclcpp sensor_msgs tf2 tf2_ros)

add_executable(vo_watchdog src/vo_watchdog.cpp)
target_include_directories(vo_watchdog PRIVATE include)
ament_target_dependencies(vo_watchdog geometry_msgs nav_msgs rclcpp rtabmap_msgs std_msgs tf2 tf2_ros)

add_executable(interface_monitor src/interface_monitor.cpp)
add_executable(depth_to_points src/depth_to_points.cpp)
add_executable(point_to_scan src/point_to_scan.cpp)
target_include_directories(depth_to_points PRIVATE include)
target_include_directories(point_to_scan PRIVATE include)
ament_target_dependencies(interface_monitor nav_msgs rclcpp sensor_msgs std_msgs)
ament_target_dependencies(depth_to_points geometry_msgs rclcpp sensor_msgs tf2 tf2_ros)
ament_target_dependencies(point_to_scan rclcpp sensor_msgs)

install(TARGETS command_guard vio_initializer mapping_odom_gate imu_conditioner path_speed_profile goal_approach_limiter dynamic_obstacle_memory vo_watchdog interface_monitor depth_to_points point_to_scan DESTINATION lib/${PROJECT_NAME})
ament_python_install_package(wla_r680_navigation_py)
install(PROGRAMS scripts/navigation_goal_bridge scripts/web_gateway scripts/furniture_semantic.py
  DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/mapping_mission scripts/prepare_mapping_run scripts/save_manual_map
  scripts/r680_nav.sh scripts/wait_mapping_ready
  scripts/resolve_navigation_map scripts/accept_navigation_map_edit
  scripts/r680_mapping.sh DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch docs web DESTINATION share/${PROJECT_NAME}
  PATTERN "__pycache__" EXCLUDE PATTERN "*.pyc" EXCLUDE)
install(FILES README.md DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  add_executable(test_relocalization_prior test/test_relocalization_prior.cpp)
  target_include_directories(test_relocalization_prior PRIVATE include)
  add_test(NAME relocalization_prior COMMAND test_relocalization_prior)
  ament_add_pytest_test(test_storage test/test_storage.py)
  ament_add_pytest_test(test_mapping_run test/test_mapping_run.py)
  add_executable(test_vio_init_policy test/test_vio_init_policy.cpp)
  target_include_directories(test_vio_init_policy PRIVATE include)
  add_test(NAME vio_init_policy COMMAND test_vio_init_policy)
  add_executable(test_isolated_clusters test/test_isolated_clusters.cpp)
  target_include_directories(test_isolated_clusters PRIVATE include)
  add_test(NAME isolated_clusters COMMAND test_isolated_clusters)
  add_executable(test_scan_projection test/test_scan_projection.cpp)
  target_include_directories(test_scan_projection PRIVATE include)
  add_test(NAME scan_projection COMMAND test_scan_projection)
endif()

ament_package()
