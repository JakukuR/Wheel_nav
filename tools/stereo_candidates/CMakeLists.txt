cmake_minimum_required(VERSION 3.22)
project(stereo_candidates LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(vpi 4.1 REQUIRED PATHS /opt/nvidia/vpi4/lib/aarch64-linux-gnu/cmake/vpi)
find_package(OpenCV REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(vpi_probe vpi_probe.cc)
target_link_libraries(vpi_probe vpi ${OpenCV_LIBS} nlohmann_json::nlohmann_json)
target_compile_options(vpi_probe PRIVATE -Wall -Wextra -Wpedantic -O2 -fno-omit-frame-pointer -ffp-contract=off)
find_package(CUDAToolkit REQUIRED)
find_path(TRT_INCLUDE NvInfer.h PATH_SUFFIXES aarch64-linux-gnu REQUIRED)
find_library(TRT_LIB nvinfer REQUIRED)
add_executable(model_probe model_probe.cc)
target_include_directories(model_probe PRIVATE ${TRT_INCLUDE})
target_link_libraries(model_probe ${OpenCV_LIBS} nlohmann_json::nlohmann_json CUDA::cudart ${TRT_LIB} ${CMAKE_DL_LIBS})
target_compile_options(model_probe PRIVATE -Wall -Wextra -Wpedantic -O2 -fno-omit-frame-pointer -ffp-contract=off)
