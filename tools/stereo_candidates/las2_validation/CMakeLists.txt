cmake_minimum_required(VERSION 3.22)
project(las2_validation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(CUDAToolkit REQUIRED)
find_package(nlohmann_json REQUIRED)
find_path(TRT_INCLUDE NvInfer.h PATH_SUFFIXES aarch64-linux-gnu REQUIRED)
find_library(TRT_LIB nvinfer REQUIRED)
add_executable(model_probe model_probe.cc)
target_include_directories(model_probe SYSTEM PRIVATE ${TRT_INCLUDE} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(model_probe PRIVATE ${OpenCV_LIBS} nlohmann_json::nlohmann_json CUDA::cudart ${TRT_LIB} ${CMAKE_DL_LIBS})
target_compile_options(model_probe PRIVATE -O2 -Wall -Wextra -Wpedantic)
